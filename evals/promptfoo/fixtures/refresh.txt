let lastSuccessfulItems = ['sample'];

export async function refresh(load: () => Promise<string[]>): Promise<string[]> {
  try {
    lastSuccessfulItems = await load();
  } catch {
    lastSuccessfulItems = [];
  }
  return lastSuccessfulItems;
}
